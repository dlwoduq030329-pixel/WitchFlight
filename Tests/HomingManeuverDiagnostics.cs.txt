            }
            File.WriteAllLines("results.csv",new[]{Header}.Concat(results.Select(r=>r.Csv)));
            var witnesses=new List<Result>();
            // Retest each far-range, rear-chase basic-craft case at all three additional rates.
            foreach(var r in results.Where(r=>r.test.broom==2&&r.test.rangeFraction==.95f&&r.test.heading==0))
            foreach(int hz in new[]{30,60,120}) {
                var c=r.test;
                witnesses.Add(Sim(new Case{broom=c.broom,hz=hz,targetFirst=c.targetFirst,rangeFraction=c.rangeFraction,heading=c.heading,trigger=c.trigger,maneuver=c.maneuver,speedMode=c.speedMode}));

            // Check representative successful patterns with finite camera swipes and reaction delay,
            // rather than assuming an instantaneous 90/180-degree mouse target change.
            var inputChecks=new List<string>{"aimRampSeconds,reactionSeconds,"+Header};
            foreach(int hz in new[]{30,60,64,120})foreach(bool order in new[]{false,true})
            foreach(float ramp in new[]{.15f,.3f})foreach(float reaction in new[]{0f,.1f,.2f})
            foreach(string mode in new[]{"boost","brake06Boost","cruise"}) {
                var c=new Case{broom=2,hz=hz,targetFirst=order,rangeFraction=.95f,heading=0,
                    trigger=mode=="boost"?25:60,maneuver=mode=="cruise"?"upThenDown":"yaw90",

            foreach(var group in results.Where(r=>r.test.rangeFraction==.95f&&r.test.heading==0&&r.test.trigger>0).GroupBy(r=>new{r.test.broom,r.test.speedMode}))
